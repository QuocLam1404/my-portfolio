@app.post("/api/search/visual")
async def visual_search(file: UploadFile):
    image = preprocess(Image.open(file.file)).unsqueeze(0)
    with torch.no_grad():
        embedding = clip_model.encode_image(image)
    results = db.products.aggregate([{
        "$vectorSearch": {
            "queryVector": embedding[0].tolist(),
            "path": "clip_embedding",
            "numCandidates": 150, "limit": 12
        }
    }])
    return {"results": list(results)}
